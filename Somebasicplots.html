EducationPanellong <- EducationPanelwide |>
  # Create a row identifier 
  mutate(row_id = row_number()) |>
  
  
  pivot_longer(
    cols = -c(1, row_id),
    names_to = c("location", "variable"),
    names_pattern = "^(.*?)\\.(.*)$" #No i do not know how to use pattern matching properly, had to look it up#
  ) |>
  
  # Pivot wider while respecting the row_id and location mapping
  pivot_wider(
    id_cols = c(row_id, location),
    names_from = variable,
    values_from = value
  ) |>
  
  # 4. Clean up the temporary row_id column
  select(-row_id) |>
    janitor::clean_names()
HousingPanellong <- HousingPanelwide |>
  # 1. Create a row identifier so each year stays separate
  mutate(row_id = row_number()) |>
  
  pivot_longer(
    cols = -c(1, row_id),
    names_to = c("location", "variable"),
    names_pattern = "^(.*?)\\.(.*)$"
  ) |>
  
  pivot_wider(
    id_cols = c(row_id, location),
    names_from = variable,
    values_from = value
  ) |>
  
  select(-row_id) |>
  janitor::clean_names()
PanelLong = merge(HousingPanellong,EducationPanellong) |>
  mutate(gentrified=ifelse(location%in%c("ancoats","quays"),1,0))
p1=ggplot(PanelLong,aes(x=year,y=graduate/(graduate+not_graduate),colour=location))+geom_line()+ylab("%Graduates")+theme_light()
p2=ggplot(PanelLong,aes(x=year,y=graduate/(graduate+not_graduate),colour=factor(gentrified)))+stat_summary(fun=mean,geom="line")+geom_jitter()+ylab("%Graduates")+theme_dark()
p2/p1
p3=ggplot(PanelLong,aes(x=year,y=local_authority_etc/(owner_occupied+other_rented+other_tenures+local_authority_etc),colour=location))+geom_line()+ylab("%LocalAuthority")+theme_light()
p4=ggplot(PanelLong,aes(x=year,y=local_authority_etc/(owner_occupied+other_rented+other_tenures+local_authority_etc),colour=factor(gentrified)))+stat_summary(fun=mean,geom="line")+geom_jitter()+ylab("%LocalAuthority")+theme_dark()
p4/p3
p5=ggplot(PanelLong,aes(x=year,y=owner_occupied/(owner_occupied+other_rented+other_tenures+local_authority_etc),colour=location))+geom_line()+ylab("%Owner_occupied")+theme_light()
p6=ggplot(PanelLong,aes(x=year,y=owner_occupied/(owner_occupied+other_rented+other_tenures+local_authority_etc),colour=factor(gentrified)))+stat_summary(fun=mean,geom="line")+geom_jitter()+ylab("%Owner_occupied")+theme_dark()
p6/p5
p7=ggplot(PanelLong,aes(x=year,y=(other_rented+other_tenures)/(owner_occupied+other_rented+other_tenures+local_authority_etc),colour=location))+geom_line()+
  ylab("%other_rented+tenures")+theme_light()
p8=ggplot(PanelLong,aes(x=year,y=(other_rented+other_tenures)/(owner_occupied+other_rented+other_tenures+local_authority_etc),colour=factor(gentrified)))+
stat_summary(fun=mean,geom="line")+geom_jitter()+ylab("%other_rented+tenures")+theme_dark()
p8/p7
