EducationPanelProp = EducationPanellong |>
  mutate(GraduateP=graduate/(graduate+not_graduate),
  NGraduateP = not_graduate/(graduate+not_graduate))
  HousingPanelProp = HousingPanellong |>
    mutate(local_authority_etcP=local_authority_etc/(owner_occupied+local_authority_etc+other_rented+other_tenures),
   owner_occupiedP=owner_occupied/(owner_occupied+local_authority_etc+other_rented+other_tenures),
    otherrented_tenuresP=(other_rented+other_tenures)/(owner_occupied+local_authority_etc+other_rented+other_tenures))
CbirthPanelProp = CbirthPanellong |>
  mutate(BornUkP=united_kingdom/(other_europe+rest_of_the_world+united_kingdom),
  BornEuropeP = other_europe/(other_europe+rest_of_the_world+united_kingdom),
  BornRestP = rest_of_the_world/(other_europe+rest_of_the_world+united_kingdom))

  EthicPanelProp = EthnicPanellong |>
  mutate(WhiteP=white/(white+black+asian+other),
  BlackP=black/(white+black+asian+other),
  AsianP=asian/(white+black+asian+other),
  OtherP=other/(white+black+asian+other))

    LongPanelProp = merge(EducationPanelProp,HousingPanelProp)|>  merge(CbirthPanelProp) |> left_join(EthicPanelProp)
    LongPanelPropj= select(LongPanelProp,c(location,year,GraduateP,local_authority_etcP,owner_occupiedP, otherrented_tenuresP,BornUkP,BornEuropeP,BornRestP,WhiteP,BlackP,AsianP,OtherP)) |>
  mutate(gentrified=ifelse(location%in%c("ancoats","quays"),1,0))
